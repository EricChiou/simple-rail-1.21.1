import com.google.gson.*;
import java.nio.file.*;
import java.nio.charset.StandardCharsets;
import java.nio.ByteBuffer;
import java.util.*;
import java.util.zip.ZipFile;

/** Non-game JSON/file audit. No Minecraft classes, registries, codecs or runtime loaded. */
public final class ResourceAudit {
    static final Gson GSON = new GsonBuilder().setPrettyPrinting().disableHtmlEscaping().create();
    static final List<Map<String,Object>> mappings = new ArrayList<>(), references = new ArrayList<>(), recipes = new ArrayList<>();
    static final List<String> failures = new ArrayList<>(), findings = new ArrayList<>();
    static final Map<String,JsonObject> models = new TreeMap<>();
    static final Map<String,Set<String>> edges = new TreeMap<>();
    static final Set<String> roots = new TreeSet<>();
    static Path resources;
    static ZipFile vanilla;
    static String id(String value) { return value.contains(":") ? value : "minecraft:"+value; }
    static String path(String id, String dir, String suffix) {
        String[] parts=id(id).split(":",2); return "assets/"+parts[0]+"/"+dir+"/"+parts[1]+suffix;
    }
    static JsonObject json(Path p) throws Exception {
        String s=Files.readString(p, StandardCharsets.UTF_8);
        return JsonParser.parseString(s.startsWith("\ufeff")?s.substring(1):s).getAsJsonObject();
    }
    static void check(boolean ok,String message){if(!ok)failures.add(message);}
    static void ref(String from,String kind,String target) {
        String canonical=id(target);
        String p=path(canonical,kind.equals("texture")?"textures":"models",kind.equals("texture")?".png":".json");
        boolean exists=canonical.equals("minecraft:builtin/generated") || canonical.equals("minecraft:builtin/entity") || Files.isRegularFile(resources.resolve(p)) || vanilla.getEntry(p)!=null;
        references.add(Map.of("from",from,"kind",kind,"id",canonical,"exists",exists));
        check(exists,"Missing "+kind+" "+canonical+" referenced by "+from);
        if(!kind.equals("texture"))edges.computeIfAbsent(from,k->new TreeSet<>()).add(canonical);
    }
    static void visit(String at,Set<String> reached,Set<String> stack){
        if(stack.contains(at)){failures.add("Model reference cycle "+at);return;}
        if(!reached.add(at))return;
        stack.add(at);
        for(String target:edges.getOrDefault(at,Set.of()))visit(target,reached,stack);
        stack.remove(at);
    }
    public static void main(String[] args)throws Exception{
        Path ev=Path.of(args[0]); resources=ev.resolve("sources/old/src/main/resources");
        JsonObject catalogue=json(Path.of("docs/evidence/T-004/catalogue.json"));
        Set<String> blocks=new TreeSet<>(),items=new TreeSet<>();
        catalogue.getAsJsonArray("blocks").forEach(x->blocks.add(x.getAsString()));
        catalogue.getAsJsonArray("items").forEach(x->items.add(x.getAsString()));
        int jsonCount=0,pngCount=0; Set<String> recipeIds=new TreeSet<>();
        Map<String,Integer> groups=new TreeMap<>(); List<Map<String,Object>> pngs=new ArrayList<>();
        try(ZipFile zip=new ZipFile(args[1])){
            vanilla=zip;
            List<Path> files;
            try(var stream=Files.walk(resources)){files=stream.filter(Files::isRegularFile).sorted().toList();}
            for(Path f:files){
                String p=resources.relativize(f).toString().replace('\\','/');
                String target=p.replace("/recipes/","/recipe/").replace("/loot_tables/","/loot_table/").replace("/tags/blocks/","/tags/block/").replace("/tags/items/","/tags/item/");
                String category=p.contains("/recipes/")?"recipe":p.contains("/loot_tables/")?"loot":p.contains("/tags/")?"tag":p.contains("/blockstates/")?"blockstate":p.contains("/models/block/")?"block model":p.contains("/models/item/")?"item model":p.contains("/textures/")?"texture":p.contains("/lang/")?"language":"metadata/logo";
                groups.merge(category,1,Integer::sum);
                String logical="not a registry object";
                if(p.startsWith("data/")){
                    String[] s=p.split("/"); int begin=category.equals("tag")?4:3;
                    logical=s[1]+":"+String.join("/",Arrays.copyOfRange(s,begin,s.length)).replaceFirst("\\.json$","");
                }else if(p.startsWith("assets/")){
                    String[] s=p.split("/"); logical=s[1]+":"+String.join("/",Arrays.copyOfRange(s,3,s.length)).replaceFirst("\\.(json|png)$","");
                }
                mappings.add(Map.of("oldPath",p,"candidatePath",target,"category",category,"logicalId",logical,"status","source reference / candidate only; not migrated"));
                if(p.endsWith(".png")){
                    byte[] bytes=Files.readAllBytes(f); pngCount++;
                    check(bytes.length>=24 && bytes[0]==(byte)137 && bytes[1]==80 && bytes[2]==78 && bytes[3]==71,"Bad PNG header "+p);
                    pngs.add(Map.of("path",p,"width",ByteBuffer.wrap(bytes,16,4).getInt(),"height",ByteBuffer.wrap(bytes,20,4).getInt()));
                }
                if(!p.endsWith(".json") && !p.endsWith(".mcmeta"))continue;
                JsonObject j=json(f); jsonCount++;
                if(category.equals("recipe")){
                    check(j.get("type").getAsString().equals("minecraft:crafting_shaped"),"Non-shaped "+p);
                    JsonArray pattern=j.getAsJsonArray("pattern"); JsonObject key=j.getAsJsonObject("key"),result=j.getAsJsonObject("result");
                    check(pattern.size()>0 && pattern.size()<=3,"Pattern height "+p);
                    Set<String> used=new TreeSet<>();int width=pattern.get(0).getAsString().length();
                    for(JsonElement row:pattern){String s=row.getAsString();check(s.length()==width && width>0 && width<=3,"Pattern width "+p);for(char c:s.toCharArray())if(c!=' ')used.add(String.valueOf(c));}
                    check(used.equals(key.keySet()),"Pattern/key mismatch "+p);
                    for(var ingredient:key.entrySet())check(ingredient.getValue().getAsJsonObject().has("item") || ingredient.getValue().getAsJsonObject().has("tag"),"Ingredient shape "+p);
                    String out=result.get("item").getAsString();int count=result.has("count")?result.get("count").getAsInt():1;
                    check(items.contains(out),"Unknown source result "+out);
                    check(count==(out.equals("simplerail:destory_rail")?3:1),"Unexpected count "+p);
                    recipeIds.add(out);
                    JsonObject candidate=j.deepCopy();candidate.getAsJsonObject("result").remove("item");candidate.getAsJsonObject("result").addProperty("id",out);
                    check(candidate.getAsJsonObject("result").get("id").getAsString().equals(out),"Candidate ID changed");
                    recipes.add(Map.of("logicalId",logical,"path",target,"result",out,"count",count,"sourcePattern",pattern,"sourceKey",key,"candidate",candidate,"validation","JSON-only; registry codec/game not executed"));
                }else if(category.equals("loot")){
                    check(j.has("type")&&j.has("pools"),"Loot structure "+p);
                    for(JsonElement pool:j.getAsJsonArray("pools")){
                        JsonObject po=pool.getAsJsonObject();check(po.has("rolls")&&po.has("entries"),"Pool structure "+p);
                        for(JsonElement entry:po.getAsJsonArray("entries")){
                            JsonObject en=entry.getAsJsonObject();check(en.get("type").getAsString().equals("minecraft:item")&&items.contains(en.get("name").getAsString()),"Loot item "+p);
                        }
                    }
                    if(logical.equals("simplerail:blocks/locomotive_cart"))findings.add("Extra loot retains blocks/locomotive_cart ID; entity type + survives_explosion requires context investigation, not a new block.");
                }else if(category.equals("tag")){
                    for(JsonElement value:j.getAsJsonArray("values")){String tagItem=value.getAsString();check((p.contains("/tags/blocks/")?blocks:items).contains(tagItem),"Tag source member "+tagItem);}
                }else if(category.equals("blockstate")){
                    String blockId="simplerail:"+f.getFileName().toString().replace(".json","");
                    String root="blockstate/"+blockId;
                    if(blocks.contains(blockId))roots.add(root);else findings.add("No source block registry entry for "+p);
                    for(var variant:j.getAsJsonObject("variants").entrySet()){
                        List<JsonElement> choices=variant.getValue().isJsonArray()?new ArrayList<>(variant.getValue().getAsJsonArray().asList()):List.of(variant.getValue());
                        for(JsonElement v:choices){JsonObject jo=v.getAsJsonObject();ref(root,"model",jo.get("model").getAsString());for(String axis:List.of("x","y"))if(jo.has(axis))check(Set.of(0,90,180,270).contains(jo.get(axis).getAsInt()),"Rotation "+p);}
                    }
                }else if(category.endsWith(" model")){
                    models.put(logical,j);
                    if(logical.startsWith("simplerail:item/")&&items.contains(logical.replace(":item/",":")))roots.add(logical);
                    if(j.has("parent"))ref(logical,"parent",j.get("parent").getAsString());
                    if(j.has("textures"))for(var texture:j.getAsJsonObject("textures").entrySet())if(!texture.getValue().getAsString().startsWith("#"))ref(logical,"texture",texture.getValue().getAsString());
                }else if(category.equals("language")){
                    for(var e:j.entrySet())check(e.getValue().isJsonPrimitive()&&e.getValue().getAsJsonPrimitive().isString(),"Language value "+p+":"+e.getKey());
                }
            }
            check(recipeIds.equals(items),"Recipe output IDs differ from source item catalogue");
            Set<String> reached=new TreeSet<>();for(String root:roots)visit(root,reached,new TreeSet<>());
            Set<String> unused=new TreeSet<>(models.keySet());unused.removeAll(reached);
            JsonObject en=json(resources.resolve("assets/simplerail/lang/en_us.json")),zh=json(resources.resolve("assets/simplerail/lang/zh_tw.json"));
            check(en.keySet().equals(zh.keySet()),"Language key mismatch");
            Map<String,Object> report=new LinkedHashMap<>();
            report.put("scope","Standalone Gson 2.10.1 / file checks. NO Minecraft codec, registry, pack reload, client/server or datagen execution.");
            report.put("resourceFiles",files.size());report.put("groups",groups);report.put("jsonParsed",jsonCount);report.put("pngHeadersChecked",pngCount);report.put("pngDimensions",pngs);report.put("languageKeysPerFile",en.size());report.put("recipes",recipes.size());report.put("referencesChecked",references.size());report.put("unreachableModelsFromRegisteredRoots",unused);report.put("findings",findings);report.put("failures",failures);report.put("exitCode",failures.isEmpty()?0:1);
            Files.writeString(ev.resolve("resource-map.json"),GSON.toJson(mappings));
            Files.writeString(ev.resolve("recipe-candidates.json"),GSON.toJson(recipes));
            Files.writeString(ev.resolve("reference-trace.json"),GSON.toJson(references));
            Files.writeString(ev.resolve("static-audit.json"),GSON.toJson(report));
            System.out.println(GSON.toJson(report));
            if(!failures.isEmpty())System.exit(1);
        }
    }
}
