function Entry-Hash($entry) {
    $stream=$entry.Open(); $hasher=[Security.Cryptography.SHA256]::Create()
    try {[BitConverter]::ToString($hasher.ComputeHash($stream)).Replace('-','')} finally {$stream.Dispose();$hasher.Dispose()}
}

function Task-Map([string]$path) {
    $map=@{}; $current=$null; $stage=-1; $stageCounts=@{}
    foreach ($line in Get-Content -LiteralPath $path -Encoding utf8) {
        $stageMatch=[regex]::Match($line,'^## 階段 ([0-8])')
        if ($stageMatch.Success) {$stage=[int]$stageMatch.Groups[1].Value; $stageCounts[$stage]=0}
        $taskMatch=[regex]::Match($line,'^### (T-[0-9]{3})')
        if ($taskMatch.Success) {
            $current=$taskMatch.Groups[1].Value; if ($map.ContainsKey($current)) {throw ('Duplicate task '+$current)}
            $map[$current]=@{heading=$line; status=''; dependencyLine=''; dependencies=@()}; $stageCounts[$stage]++
        } elseif ($line.StartsWith('## ')) {$current=$null}
        if ($null -ne $current) {
            if ($line.StartsWith('- 狀態：')) {$map[$current].status=$line.Substring(5)}
            if ($line.StartsWith('- 前置任務')) {$map[$current].dependencyLine=$line; $map[$current].dependencies=@([regex]::Matches($line,'T-[0-9]{3}') | ForEach-Object {$_.Value})}
        }
    }
    return @{tasks=$map; stages=$stageCounts}
}

function Coverage-Text([string]$path) {
    $collect=$false; $lines=[Collections.Generic.List[string]]::new()
    foreach ($line in Get-Content -LiteralPath $path -Encoding utf8) {
        if ($line.StartsWith('## ') -and $line.Contains('覆蓋')) {$collect=$true} elseif ($collect -and $line.StartsWith('## ')) {break}
        if ($collect) {$lines.Add($line)}
    }
    return ($lines -join "`n").Trim()
}
